cmake_minimum_required(VERSION 3.22)

if(NOT DEFINED IDF_TARGET)
    set(IDF_TARGET esp32s3)
endif()
if(NOT IDF_TARGET STREQUAL "esp32s3")
    message(FATAL_ERROR "WristRelay-S3 requires the ESP32-S3 target")
endif()

set(EXTRA_COMPONENT_DIRS firmware_core)
option(WR_WAKE_WAKENET "Build optional on-device WakeNet detection" OFF)
if(WR_WAKE_WAKENET)
    list(APPEND EXTRA_COMPONENT_DIRS wake_profile)
    if(NOT DEFINED SDKCONFIG_DEFAULTS)
        set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;sdkconfig.wakenet")
    endif()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
if(WR_WAKE_WAKENET)
    idf_build_set_property(DEPENDENCIES_LOCK dependencies.lock.wakenet)
endif()
project(wristrelay_s3)
